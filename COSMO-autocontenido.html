3. Determinación del centro de presión (CP)

Método algorítmico (Python)

ProgramaParaCalcularCP.py
# ==========================================
# 2. Variables con las dimensiones del cohete
# ==========================================
alto_cono = 9.0       
ancho_cono =3.5       
largo_cuerpo = 31.5    

cuerda_aleta = 5.0    
ancho_aleta = 2.5      
numero_aletas = 4      

factor_influencia = 1 + (numero_aletas * ancho_aleta) / (ancho_cono * 3)

area_cono = (ancho_cono * alto_cono) / 2
area_cuerpo = ancho_cono * largo_cuerpo

area_una_aleta = cuerda_aleta * ancho_aleta
area_aletas_total = area_una_aleta * numero_aletas

area_aletas_efectiva = area_aletas_total * factor_influencia

# 5. Ubicación de los Centroides (medidos desde la punta)
x_cono = (2 / 3) * alto_cono
x_cuerpo = alto_cono + (largo_cuerpo / 2)
x_aletas = alto_cono + largo_cuerpo - cuerda_aleta/2

# ==========================================
numerador = (area_cono * x_cono) + (area_cuerpo * x_cuerpo) + (area_aletas_efectiva * x_aletas)
denominador = area_cono + area_cuerpo + area_aletas_efectiva

cp = numerador / denominador

# ==========================================
# 7. Impresión del valor final
# ==========================================
print(f"El Centro de Presión (CP) es: {cp:.2f}")
